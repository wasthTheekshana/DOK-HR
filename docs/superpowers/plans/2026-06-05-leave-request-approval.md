# Leave Request & Approval Flow Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a standalone Leave module where staff submit leave requests, supervisors approve/reject for their sites, and admins can see all requests and override any decision.

**Architecture:** New `leave_requests` PostgreSQL table (migration in `db/config.ts`), new `leaveController.ts` + `leaveRoutes.ts` on the server, new `Leave.tsx` page on the client. Follows the exact same patterns as `attendanceController.ts` (named-param `execute()` calls, role-filtered queries) and `Layout.tsx` nav structure (`NAV_GROUPS` array). Purely informational — no payroll impact.

**Tech Stack:** PostgreSQL, Express + TypeScript, React 18 + TypeScript + Tailwind CSS, Lucide React icons, react-hot-toast.

```typescript
import { Response } from 'express';
import { execute } from '../db/dbUtils';
import { AuthRequest } from '../middleware/authMiddleware';

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

export const getLeaveRequests = async (req: AuthRequest, res: Response) => {
    const callerId = req.user?.id;
    const callerRole = req.user?.role;
    const { status, site_id } = req.query;

    try {
        let query = `
            SELECT lr.id, lr.staff_id, u.name AS staff_name,
                   lr.site_id, s.site_no, s.name AS site_name,
                   TO_CHAR(lr.date_from, 'YYYY-MM-DD') AS date_from,
                   TO_CHAR(lr.date_to,   'YYYY-MM-DD') AS date_to,
                   lr.reason, lr.status,
                   lr.reviewed_by, rv.name AS reviewer_name,
                   TO_CHAR(lr.reviewed_at, 'YYYY-MM-DD"T"HH24:MI:SS') AS reviewed_at,
                   lr.admin_note,
                   TO_CHAR(lr.created_at, 'YYYY-MM-DD"T"HH24:MI:SS') AS created_at
            FROM leave_requests lr
            JOIN users u ON lr.staff_id = u.id
            JOIN sites s ON lr.site_id  = s.id
            LEFT JOIN users rv ON lr.reviewed_by = rv.id
            WHERE 1=1
        `;
        const params: any = {};

        if (callerRole === 'staff') {
            query += ` AND lr.staff_id = :callerId`;
            params.callerId = callerId;
        } else if (callerRole === 'supervisor') {
            query += ` AND lr.site_id IN (SELECT id FROM sites WHERE supervisor_id = :callerId)`;
            params.callerId = callerId;
        } else {
            // admin / system_admin — optional filters
            if (site_id) {
                query += ` AND lr.site_id = :site_id`;
                params.site_id = String(site_id);
            }
        }

        if (status && ['pending', 'approved', 'rejected'].includes(String(status))) {
            query += ` AND lr.status = :status`;
            params.status = String(status);
        }

        query += ` ORDER BY lr.created_at DESC`;

        const result = await execute<any>(query, params);
        res.json(result.rows || []);
    } catch (err) {
        console.error('getLeaveRequests error:', err);
        res.status(500).json({ message: 'Server error' });
    }
};

export const createLeaveRequest = async (req: AuthRequest, res: Response) => {
    const staffId = req.user?.id;
    const siteId  = req.user?.site_id;
    const { date_from, date_to, reason } = req.body;

    if (!date_from || !date_to) {
        return res.status(400).json({ message: 'date_from and date_to are required.' });
    }
    if (!DATE_RE.test(String(date_from)) || !DATE_RE.test(String(date_to))) {
        return res.status(400).json({ message: 'Dates must be in YYYY-MM-DD format.' });
    }
    if (String(date_from) > String(date_to)) {
        return res.status(400).json({ message: 'date_from must not be after date_to.' });
    }
    if (!siteId) {
        return res.status(400).json({ message: 'You are not assigned to a site.' });
    }
    if (reason && String(reason).length > 500) {
        return res.status(400).json({ message: 'Reason must be 500 characters or less.' });
    }

    try {
        const result = await execute<any>(
            `INSERT INTO leave_requests (staff_id, site_id, date_from, date_to, reason)
             VALUES (:staff_id, :site_id, :date_from, :date_to, :reason)
             RETURNING id`,
            { staff_id: staffId, site_id: siteId, date_from: String(date_from), date_to: String(date_to), reason: reason || null }
        );
        const id = result.rows?.[0]?.ID ?? result.rows?.[0]?.id;
        res.status(201).json({ id, message: 'Leave request submitted.' });
    } catch (err) {
        console.error('createLeaveRequest error:', err);
        res.status(500).json({ message: 'Server error' });
    }
};

export const reviewLeaveRequest = async (req: AuthRequest, res: Response) => {
    const callerId   = req.user?.id;
    const callerRole = req.user?.role;
    const { id } = req.params;
    const { status, admin_note } = req.body;

    if (!['approved', 'rejected'].includes(String(status))) {
        return res.status(400).json({ message: 'status must be approved or rejected.' });
    }
    if (isNaN(Number(id))) {
        return res.status(400).json({ message: 'Invalid id.' });
    }

    try {
        const found = await execute<any>(
            `SELECT lr.id, lr.site_id FROM leave_requests lr WHERE lr.id = :id`,
            { id: Number(id) }
        );
        if (!found.rows || found.rows.length === 0) {
            return res.status(404).json({ message: 'Leave request not found.' });
        }
        const siteId = found.rows[0].SITE_ID ?? found.rows[0].site_id;

        if (callerRole === 'supervisor') {
            const owns = await execute<any>(
                `SELECT id FROM sites WHERE id = :site_id AND supervisor_id = :callerId`,
                { site_id: siteId, callerId }
            );
            if (!owns.rows || owns.rows.length === 0) {
                return res.status(403).json({ message: 'You do not supervise this site.' });
            }
        }

        await execute(
            `UPDATE leave_requests
             SET status = :status, reviewed_by = :reviewed_by, reviewed_at = NOW(), admin_note = :admin_note
             WHERE id = :id`,
            { status: String(status), reviewed_by: callerId, admin_note: admin_note || null, id: Number(id) }
        );
        res.json({ message: `Leave request ${status}.` });
    } catch (err) {
        console.error('reviewLeaveRequest error:', err);
        res.status(500).json({ message: 'Server error' });
    }
};

export const overrideLeaveRequest = async (req: AuthRequest, res: Response) => {
    const callerId = req.user?.id;
    const { id } = req.params;
    const { status, admin_note } = req.body;

    if (!['approved', 'rejected'].includes(String(status))) {
        return res.status(400).json({ message: 'status must be approved or rejected.' });
    }
    if (isNaN(Number(id))) {
        return res.status(400).json({ message: 'Invalid id.' });
    }

    try {
        const found = await execute<any>(
            `SELECT id FROM leave_requests WHERE id = :id`,
            { id: Number(id) }
        );
        if (!found.rows || found.rows.length === 0) {
            return res.status(404).json({ message: 'Leave request not found.' });
        }

        await execute(
            `UPDATE leave_requests
             SET status = :status, reviewed_by = :reviewed_by, reviewed_at = NOW(), admin_note = :admin_note
             WHERE id = :id`,
            { status: String(status), reviewed_by: callerId, admin_note: admin_note || null, id: Number(id) }
        );
        res.json({ message: `Leave request overridden to ${status}.` });
    } catch (err) {
        console.error('overrideLeaveRequest error:', err);
        res.status(500).json({ message: 'Server error' });
    }
};
```

```tsx
import React, { useEffect, useState } from 'react';
import api from '../services/api';
import { useAuth } from '../context/AuthContext';
import type { LeaveRequest, Site } from '../types';
import toast from 'react-hot-toast';

const StatusBadge: React.FC<{ status: string }> = ({ status }) => {
    const cls =
        status === 'approved' ? 'bg-green-100 text-green-800' :
        status === 'rejected' ? 'bg-red-100 text-red-800' :
        'bg-yellow-100 text-yellow-800';
    return <span className={`inline-flex items-center px-2 py-0.5 rounded-full text-xs font-semibold capitalize ${cls}`}>{status}</span>;
};

const LeavePage: React.FC = () => {
    const { role } = useAuth();
    const isStaff      = role === 'staff';
    const isSupervisor = role === 'supervisor';
    const isAdmin      = role === 'admin' || role === 'system_admin';

    const [loading, setLoading]     = useState(true);
    const [refreshing, setRefreshing] = useState(false);
    const [tab, setTab]             = useState<'pending' | 'all'>('pending');

    // Admin filters
    const [sites, setSites]         = useState<Site[]>([]);
    const [filterSite, setFilterSite]   = useState('');
    const [filterStatus, setFilterStatus] = useState('');

    const [newFrom, setNewFrom]     = useState('');
    const [newTo, setNewTo]         = useState('');
    const [newReason, setNewReason] = useState('');
    const [newError, setNewError]   = useState('');
    const [submitting, setSubmitting] = useState(false);

    // Review modal
    const [reviewTarget, setReviewTarget] = useState<{ id: number; action: ReviewAction } | null>(null);
    const [reviewNote, setReviewNote]     = useState('');
    const [reviewing, setReviewing]       = useState(false);

    const fetchRequests = async (quiet = false) => {
        if (quiet) setRefreshing(true);
        else setLoading(true);
        try {
            const params: any = {};
            if (isAdmin) {
                if (filterSite)   params.site_id = filterSite;
                if (filterStatus) params.status  = filterStatus;
            }
            const res = await api.get('/leave', { params });
            setRequests(res.data);
        } catch {
            toast.error('Failed to load leave requests.');
        } finally {
            setLoading(false);
            setRefreshing(false);
        }

        fetchRequests();
    }, [filterSite, filterStatus]);

        if (!isAdmin) return;
        api.get('/sites').then(r => setSites(r.data)).catch(() => {});
    }, [isAdmin]);

        e.preventDefault();
        setNewError('');
        if (!newFrom || !newTo) { setNewError('Both dates are required.'); return; }
        if (newFrom > newTo)    { setNewError('Start date must not be after end date.'); return; }
        setSubmitting(true);
        try {
            await api.post('/leave', { date_from: newFrom, date_to: newTo, reason: newReason || undefined });
            toast.success('Leave request submitted.');
            setShowNew(false);
            setNewFrom(''); setNewTo(''); setNewReason('');
            fetchRequests(true);
        } catch (err: any) {
            toast.error(err.response?.data?.message || 'Failed to submit request.');

            setSubmitting(false);

        setReviewNote('');

        try {
            await api.put(endpoint, { status, admin_note: reviewNote || undefined });
            toast.success(`Leave request ${status}.`);
            setReviewTarget(null);
            fetchRequests(true);
        } catch (err: any) {
            toast.error(err.response?.data?.message || 'Failed to update request.');
        } finally {
            setReviewing(false);
        }

            {/* Header */}

                <div className="flex items-center gap-3">

                            {isStaff ? 'Submit and track your leave requests' :
                             isSupervisor ? 'Review leave requests from your team' :

                <div className="flex items-center gap-2">
                    <button onClick={() => fetchRequests(true)} disabled={refreshing}
                        className="p-2 text-slate-400 hover:text-slate-700 hover:bg-slate-100 rounded-lg transition-colors">
                        <RefreshCw className={`w-4 h-4 ${refreshing ? 'animate-spin' : ''}`} />
                    </button>
                    {isStaff && (
                        <button onClick={() => setShowNew(true)}
                            className="flex items-center gap-1.5 px-3 py-2 bg-indigo-600 text-white text-sm font-semibold rounded-lg hover:bg-indigo-700 transition-colors">
                            <Plus className="w-4 h-4" />
                            New Request
                        </button>
                    )}
                </div>
            </div>

            {/* Admin filters */}
            {isAdmin && (
                <div className="flex flex-wrap gap-2">
                    <select value={filterSite} onChange={e => setFilterSite(e.target.value)}
                        className="form-input w-auto text-sm">
                        <option value="">All Sites</option>
                        {sites.map(s => <option key={s.ID} value={String(s.ID)}>{s.SITE_NO} — {s.NAME}</option>)}
                    </select>
                    <select value={filterStatus} onChange={e => setFilterStatus(e.target.value)}
                        className="form-input w-auto text-sm">
                        <option value="">All Statuses</option>
                        <option value="pending">Pending</option>
                        <option value="approved">Approved</option>
                        <option value="rejected">Rejected</option>
                    </select>
                </div>

            {/* Tabs (supervisor + admin when no filter applied) */}

                                    <th className="text-left px-4 py-2.5 text-xs font-semibold text-slate-500 uppercase tracking-wider">Status</th>
                                    {(isSupervisor || isAdmin) && <th className="text-left px-4 py-2.5 text-xs font-semibold text-slate-500 uppercase tracking-wider">Reviewed By</th>}
                                    {isStaff && <th className="text-left px-4 py-2.5 text-xs font-semibold text-slate-500 uppercase tracking-wider">Note</th>}
                                    {(isSupervisor || isAdmin) && <th className="px-4 py-2.5 text-xs font-semibold text-slate-500 uppercase tracking-wider text-right">Actions</th>}
                                </tr>
                            </thead>
                            <tbody className="divide-y divide-slate-100">
                                {displayed.map(r => (
                                    <tr key={r.ID} className="hover:bg-slate-50/50 transition-colors">
                                        {!isStaff && <td className="px-4 py-3 font-medium text-slate-800">{r.STAFF_NAME}</td>}
                                        {!isStaff && <td className="px-4 py-3 text-slate-500 text-xs">{r.SITE_NO}</td>}
                                        <td className="px-4 py-3 text-slate-700 whitespace-nowrap">
                                            {r.DATE_FROM === r.DATE_TO ? r.DATE_FROM : `${r.DATE_FROM} → ${r.DATE_TO}`}
                                        </td>
                                        <td className="px-4 py-3 text-slate-600 max-w-[200px] truncate" title={r.REASON || undefined}>
                                            {r.REASON || <span className="text-slate-300 italic">—</span>}
                                        </td>
                                        <td className="px-4 py-3"><StatusBadge status={r.STATUS} /></td>
                                        {(isSupervisor || isAdmin) && (
                                            <td className="px-4 py-3 text-slate-500 text-xs">
                                                {r.REVIEWER_NAME || <span className="text-slate-300">—</span>}
                                            </td>
                                        )}
                                        {isStaff && (
                                            <td className="px-4 py-3 text-slate-500 text-xs max-w-[160px] truncate" title={r.ADMIN_NOTE || undefined}>
                                                {r.ADMIN_NOTE || <span className="text-slate-300">—</span>}
                                            </td>
                                        )}
                                        {(isSupervisor || isAdmin) && (
                                            <td className="px-4 py-3 text-right">
                                                {r.STATUS === 'pending' ? (
                                                    <div className="flex justify-end gap-1.5">
                                                        <button onClick={() => openReview(r.ID, 'approve')}
                                                            title="Approve"
                                                            className="p-1.5 text-green-600 hover:bg-green-50 rounded-lg transition-colors">
                                                            <CheckCircle className="w-4 h-4" />
                                                        </button>
                                                        <button onClick={() => openReview(r.ID, 'reject')}
                                                            title="Reject"
                                                            className="p-1.5 text-red-500 hover:bg-red-50 rounded-lg transition-colors">
                                                            <XCircle className="w-4 h-4" />
                                                        </button>
                                                    </div>
                                                ) : isAdmin ? (
                                                    <button onClick={() => openReview(r.ID, r.STATUS === 'approved' ? 'override_reject' : 'override_approve')}
                                                        title="Override decision"
                                                        className="flex items-center gap-1 px-2 py-1 text-xs text-slate-500 hover:text-slate-700 hover:bg-slate-100 rounded-lg transition-colors">
                                                        <RotateCcw className="w-3 h-3" />
                                                        Override
                                                    </button>
                                                ) : (
                                                    <span className="text-slate-300 text-xs">—</span>
                                                )}
                                            </td>
                                        )}
                                    </tr>
                                ))}
                            </tbody>
                        </table>
                    </div>
                )}
            </div>

            {/* New Request Modal */}
            {showNew && (
                <div className="modal-overlay" onClick={() => setShowNew(false)}>
                    <div className="bg-white rounded-2xl shadow-xl w-full max-w-[420px] overflow-hidden" onClick={e => e.stopPropagation()}>
                        <div className="flex items-center justify-between px-6 py-4 border-b border-slate-100">
                            <h2 className="text-[15px] font-bold text-slate-900">New Leave Request</h2>
                            <button onClick={() => setShowNew(false)} className="p-2 text-slate-400 hover:text-slate-600 hover:bg-slate-100 rounded-lg transition-colors">
                                <X className="w-4 h-4" />
                            </button>
                        </div>
                        <form onSubmit={handleSubmitNew} className="p-6 space-y-4">
                            <div className="grid grid-cols-2 gap-3">
                                <div>
                                    <label className="form-label">From</label>
                                    <input type="date" value={newFrom} onChange={e => setNewFrom(e.target.value)}
                                        className="form-input" required />
                                </div>
                                <div>
                                    <label className="form-label">To</label>
                                    <input type="date" value={newTo} onChange={e => setNewTo(e.target.value)}
                                        className="form-input" required />
                                </div>
                            </div>
                            <div>
                                <label className="form-label">Reason <span className="text-slate-400 font-normal">(optional)</span></label>
                                <textarea value={newReason} onChange={e => setNewReason(e.target.value)}
                                    maxLength={500} rows={3}
                                    className="form-input resize-none"
                                    placeholder="Briefly describe the reason..." />
                                <p className="text-xs text-slate-400 mt-1 text-right">{newReason.length}/500</p>
                            </div>
                            {newError && <p className="text-sm text-red-600 font-medium">{newError}</p>}
                            <div className="flex gap-2.5 pt-1">
                                <button type="button" onClick={() => setShowNew(false)} className="btn btn-ghost flex-1">Cancel</button>
                                <button type="submit" disabled={submitting} className="btn btn-primary flex-1">
                                    {submitting && <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />}
                                    Submit
                                </button>
                            </div>
                        </form>
                    </div>
                </div>
            )}

            {/* Review / Override Modal */}
            {reviewTarget && (
                <div className="modal-overlay" onClick={() => setReviewTarget(null)}>
                    <div className="bg-white rounded-2xl shadow-xl w-full max-w-[380px] overflow-hidden" onClick={e => e.stopPropagation()}>
                        <div className="flex items-center justify-between px-6 py-4 border-b border-slate-100">
                            <h2 className="text-[15px] font-bold text-slate-900">
                                {reviewTarget.action === 'approve' ? 'Approve Leave' :
                                 reviewTarget.action === 'reject'  ? 'Reject Leave' :
                                 reviewTarget.action === 'override_approve' ? 'Override: Approve' :
                                 'Override: Reject'}
                            </h2>
                            <button onClick={() => setReviewTarget(null)} className="p-2 text-slate-400 hover:text-slate-600 hover:bg-slate-100 rounded-lg transition-colors">
                                <X className="w-4 h-4" />
                            </button>
                        </div>
                        <div className="p-6 space-y-4">
                            <div>
                                <label className="form-label">Note <span className="text-slate-400 font-normal">(optional)</span></label>
                                <input type="text" value={reviewNote} onChange={e => setReviewNote(e.target.value)}
                                    maxLength={300} className="form-input"
                                    placeholder="Add a note for the staff member..." />
                            </div>
                            <div className="flex gap-2.5 pt-1">
                                <button type="button" onClick={() => setReviewTarget(null)} className="btn btn-ghost flex-1">Cancel</button>
                                <button onClick={handleReview} disabled={reviewing}
                                    className={`btn flex-1 text-white ${
                                        reviewTarget.action.includes('approve') ? 'bg-green-600 hover:bg-green-700' : 'bg-red-600 hover:bg-red-700'
                                    }`}>
                                    {reviewing && <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />}
                                    Confirm
                                </button>
                            </div>
                        </div>
                    </div>
                </div>
            )}
        </div>
    );
};

export default LeavePage;
```
